import pandas as pd
from fastavro import writer, parse_schema

# Load dataset
df = pd.read_csv("customers.csv")
print("Dataset Loaded Successfully")
print("Dataset Shape:", df.shape)

# 1. Convert to JSON
df.to_json(
    "customers.json",
    orient="records",
    indent=4
)
print("JSON file created successfully")

# 2. Convert to Parquet
df.to_parquet(
    "customers.parquet",
    index=False
)
print("Parquet file created successfully")

# 3. Convert to ORC
df.to_orc(
    "customers.orc",
    index=False
)
print("ORC file created successfully")

# 4. Convert to Avro
# Use the actual column names from the dataset
schema = {
    "type": "record",
    "name": "Customer",
    "fields": [
        {
            "name": str(col),
            "type": ["null", "string"]
        }
        for col in df.columns
    ]
}

parsed_schema = parse_schema(schema)

# Convert all values to strings and handle missing values
records = []
for record in df.to_dict(orient="records"):
    clean_record = {}
    for col, value in record.items():
        clean_record[str(col)] = (
            None if pd.isna(value) else str(value)
        )
    records.append(clean_record)

with open("customers.avro", "wb") as out:
    writer(out, parsed_schema, records)

print("Avro file created successfully")
print("\nAll four file formats generated successfully!")

import os
import pandas as pd

# List of files
files = [
    "customers.csv",
    "customers.json",
    "customers.parquet",
    "customers.orc",
    "customers.avro"
]

# 1. File size comparison
print("=" * 50)
print("FILE SIZE COMPARISON")
print("=" * 50)

for file in files:
    size = os.path.getsize(file) / (1024 * 1024)
    print(f"{file}: {size:.4f} MB")

# 2. Store file sizes in a DataFrame
storage_data = []

for file in files:
    size = round(os.path.getsize(file) / (1024 * 1024), 4)
    storage_data.append([file, size])

storage_df = pd.DataFrame(
    storage_data,
    columns=["Format", "Size (MB)"]
)

print("\nSTORAGE COMPARISON TABLE")
print(storage_df)

# 3. Calculate storage reduction compared with CSV
csv_size = os.path.getsize("customers.csv")

print("\nSTORAGE REDUCTION COMPARED WITH CSV")

for file in files:
    if file == "customers.csv":
        continue

    size = os.path.getsize(file)

    reduction = ((csv_size - size) / csv_size) * 100

    print(
        file,
        "Storage Reduction:",
        round(reduction, 2),
        "%"
    )

# 4. Save comparison results
storage_df.to_csv("storage_comparison.csv", index=False)

print("\nStorage comparison report saved successfully")

import pandas as pd
import time
from fastavro import reader
import matplotlib.pyplot as plt

benchmark_results = []

# 1. CSV read time
start = time.perf_counter()
csv_df = pd.read_csv("customers.csv")
end = time.perf_counter()

benchmark_results.append({
    "Format": "CSV",
    "Read_Time_Seconds": round(end - start, 5),
    "Rows_Read": csv_df.shape[0],
    "Columns_Read": csv_df.shape[1]
})

# 2. JSON read time
start = time.perf_counter()
json_df = pd.read_json("customers.json")
end = time.perf_counter()

benchmark_results.append({
    "Format": "JSON",
    "Read_Time_Seconds": round(end - start, 5),
    "Rows_Read": json_df.shape[0],
    "Columns_Read": json_df.shape[1]
})

# 3. Parquet read time
start = time.perf_counter()
parquet_df = pd.read_parquet("customers.parquet")
end = time.perf_counter()

benchmark_results.append({
    "Format": "Parquet",
    "Read_Time_Seconds": round(end - start, 5),
    "Rows_Read": parquet_df.shape[0],
    "Columns_Read": parquet_df.shape[1]
})

# 4. ORC read time
start = time.perf_counter()
orc_df = pd.read_orc("customers.orc")
end = time.perf_counter()

benchmark_results.append({
    "Format": "ORC",
    "Read_Time_Seconds": round(end - start, 5),
    "Rows_Read": orc_df.shape[0],
    "Columns_Read": orc_df.shape[1]
})

# 5. Avro read time
start = time.perf_counter()

with open("customers.avro", "rb") as file:
    avro_records = list(reader(file))

avro_df = pd.DataFrame(avro_records)
end = time.perf_counter()

benchmark_results.append({
    "Format": "Avro",
    "Read_Time_Seconds": round(end - start, 5),
    "Rows_Read": avro_df.shape[0],
    "Columns_Read": avro_df.shape[1]
})

# Display benchmark results
benchmark_df = pd.DataFrame(benchmark_results)

print("=" * 50)
print("READ PERFORMANCE BENCHMARK")
print("=" * 50)
print(benchmark_df.to_string(index=False))

# Identify the fastest format
fastest = benchmark_df.loc[
    benchmark_df["Read_Time_Seconds"].idxmin()
]

print("\nFASTEST FORMAT:", fastest["Format"])
print("READ TIME:", fastest["Read_Time_Seconds"], "seconds")

# Save results
benchmark_df.to_csv("benchmark_results.csv", index=False)
print("\nBenchmark results saved successfully")

# Plot read-time comparison
plt.figure(figsize=(9, 5))
plt.bar(
    benchmark_df["Format"],
    benchmark_df["Read_Time_Seconds"]
)
plt.xlabel("File Format")
plt.ylabel("Read Time (Seconds)")
plt.title("File Format Read Performance Comparison")
plt.tight_layout()
plt.savefig("benchmark_chart.png")
plt.show()
